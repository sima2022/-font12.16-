void drawKanji12(uint16_t x, uint16_t y, uint16_t jis,
                 uint16_t fg, uint16_t bg)
{
    uint16_t idx = jis - SHINONOME_JIS_START;
    const uint8_t *font = shinonome_kanji_lv1_12[idx];

    for (int row = 0; row < 12; row++) {

        uint8_t upper = font[row];              // 上位8bit
        uint8_t lower = font[12 + row / 2];     // 下位4bit（2行で1バイト）

        // 偶数行 → 上位4bit、奇数行 → 下位4bit
        if (row & 1)
            lower &= 0x0F;          // 下位4bit
        else
            lower >>= 4;            // 上位4bit

        // 12bit のビット列を構成
        uint16_t bits = ((uint16_t)upper << 4) | lower;

        for (int col = 0; col < 12; col++) {
            if (bits & (0x800 >> col))
                drawPixel(x + col, y + row, fg);
            else
                drawPixel(x + col, y + row, bg);
        }
    }
}
void drawKanji16(uint16_t x, uint16_t y, uint16_t jis, uint16_t fg, uint16_t bg)
{
    uint16_t idx = jis - SHINONOME_JIS_START;
    const uint8_t *font = shinonome_kanji_lv1_16[idx];

    for (int row = 0; row < 16; row++) {
        uint16_t bits = (font[row*2] << 8) | font[row*2 + 1];
        for (int col = 0; col < 16; col++) {
            if (bits & (0x8000 >> col)) {
                drawPixel(x + col, y + row, fg);
            } else {
                drawPixel(x + col, y + row, bg);
            }
        }
    }
}
